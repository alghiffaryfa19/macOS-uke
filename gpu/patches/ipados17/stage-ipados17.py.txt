#!/usr/bin/env python3
"""Stage the Adreno 732 kernel sources for the macOS 26.5 (ForSheng) kernel into boot/macos-gpu/src.

The iPadOS 17 kernel that ran Metal on this GPU (metal_up2, UniXNU-uke-kbuild branch uke-gpu) built the driver from
three places; this copies exactly those, the same way boot/macos-touch/src holds the touch sources:

  src/drivers/NativeAdreno   UniXNU-uke-kbuild/drivers/NativeAdreno (*.cpp, *.h; not tests/)
  src/package                UniXNU-uke-kbuild/package: GpuPowerProbe.cpp, GpuPowerBoard.h, ShengGmu*.{c,h}
  src/kits/<kit>             UniXNU-core-kbuild/kits/{AdrenoFirmware,ArmSmmu4K,AdrenoPM4} (*.c, *.h)
  src/generated              ShengGmu{,Votes,CommandDb}Source.h from scripts/prepare-uke-gmu.py's build mode
  src/firmware.S             gen71100 GMU/SQE/ZAP as the strong native_adreno_*_image symbols (private: never commit)

Then patches/NNNN-*.patch are applied to the copies. Our own macOS glue lives in src/macos (UkeGpu.cpp,
UkeGpuFirmwareRemap.h) and is not overwritten. Nothing in the source repositories changes.
Needs Python 3.11+ (tomllib), e.g. refs/macOS-ForSheng/.venv/bin/python.
Usage: refs/macOS-ForSheng/.venv/bin/python -I boot/macos-gpu/stage.py [--no-firmware]
"""
import argparse
import hashlib
import json
import runpy
import shutil
import subprocess
import sys
from pathlib import Path

D = Path(__file__).resolve().parents[2]
HERE = Path(__file__).resolve().parent
DEVICE = D / 'UniXNU-uke-kbuild'
CORE = D / 'UniXNU-core-kbuild'
SRC = HERE / 'src'
PACKAGE = ['GpuPowerProbe.cpp', 'GpuPowerBoard.h', 'ShengGmu.c', 'ShengGmu.h', 'ShengGmuVotes.c', 'ShengGmuVotes.h',
           'ShengGmuCommandDb.c', 'ShengGmuCommandDb.h']
KITS = ['AdrenoFirmware', 'ArmSmmu4K', 'AdrenoPM4']
FIRMWARE = [('_native_adreno_firmware_image', 'gmu'), ('_native_adreno_sqe_firmware_image', 'sqe'),
            ('_native_adreno_zap_image', 'zap')]


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def copy_tree(source, destination, patterns):
    if destination.exists():
        shutil.rmtree(destination)
    destination.mkdir(parents=True)
    copied = {}
    for pattern in patterns:
        for path in sorted(source.glob(pattern)):
            if path.is_file():
                shutil.copy2(path, destination / path.name)
                copied[path.name] = sha(path)
    return copied


def generate_gmu_headers(out):
    # prepare-uke-gmu.py writes to <repo>/build/sheng-gmu; run its build mode with OUT pointed here instead.
    module = runpy.run_path(str(DEVICE / 'scripts/prepare-uke-gmu.py'), run_name='uke_gmu')
    build = module['build']
    build.__globals__['OUT'] = out
    build(argparse.Namespace(force=True))
    return {name: sha(out / name) for name in ('ShengGmuSource.h', 'ShengGmuVotesSource.h', 'ShengGmuCommandDbSource.h')}


def firmware_assembly(out):
    inputs = json.loads((DEVICE / 'scripts/data/uke-gpu-inputs.json').read_text())['firmware']
    lines = ['.section __DATA_CONST,__const']
    pins = {}
    for label, role in FIRMWARE:
        pin = inputs[role]
        path = DEVICE / '.cache/firmware/uke-gpu' / pin['name']
        blob = path.read_bytes()
        if len(blob) != pin['bytes'] or hashlib.sha256(blob).hexdigest() != pin['sha256']:
            sys.exit('firmware %s does not match its recorded pin' % pin['name'])
        lines += ['.p2align 12', '.globl ' + label, label + ':', '.incbin ' + json.dumps(str(path)), label + '_end:',
                  '.p2align 3', '.globl ' + label + '_size', label + '_size:', '.quad ' + label + '_end-' + label]
        pins[pin['name']] = dict(bytes=len(blob), sha256=pin['sha256'])
    (out / 'firmware.S').write_text('\n'.join(lines) + '\n')
    return pins


def main():
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument('--no-firmware', action='store_true', help='leave the weak empty firmware definitions in place')
    args = parser.parse_args()
    record = dict(device_repo=str(DEVICE), core_repo=str(CORE))
    record['native_adreno'] = copy_tree(DEVICE / 'drivers/NativeAdreno', SRC / 'drivers/NativeAdreno', ['*.cpp', '*.h'])
    package = SRC / 'package'
    if package.exists():
        shutil.rmtree(package)
    package.mkdir(parents=True)
    record['package'] = {}
    for name in PACKAGE:
        shutil.copy2(DEVICE / 'package' / name, package / name)
        record['package'][name] = sha(package / name)
    record['kits'] = {kit: copy_tree(CORE / 'kits' / kit, SRC / 'kits' / kit, ['*.c', '*.h']) for kit in KITS}
    generated = SRC / 'generated'
    generated.mkdir(parents=True, exist_ok=True)
    record['generated'] = generate_gmu_headers(generated)
    # Our macOS changes to the copied iPadOS sources (patches/NNNN-*.patch, in order); NativeUfs patches are not ours to apply.
    record['patches'] = []
    for patch in sorted((HERE / 'patches').glob('[0-9][0-9][0-9][0-9]-*.patch')):
        subprocess.run(['git', 'apply', '--whitespace=nowarn', str(patch)], cwd=SRC, check=True)
        record['patches'].append(patch.name)
    (SRC / 'firmware.S').unlink(missing_ok=True)
    if not args.no_firmware:
        record['firmware'] = firmware_assembly(SRC)
    (HERE / 'stage.json').write_text(json.dumps(record, indent=2) + '\n')
    print('staged %d NativeAdreno files, %d package files, kits %s, firmware %s' % (
        len(record['native_adreno']), len(record['package']), ', '.join(KITS),
        'embedded' if not args.no_firmware else 'absent (weak)'))


if __name__ == '__main__':
    main()
